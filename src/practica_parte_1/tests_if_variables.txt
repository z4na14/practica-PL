int global = 100;
int resultado = 0;
int a = 0;
int b = 0;
int main() {
    // prueba 1 - if sin else con variable local
    int x = 10;
    if (x == 10) {
        x = 99;
    }
    printf("%d", x);

    // prueba 2 - if con else
    int y = 5;
    if (y == 0) {
        b = 123;
    } else {
        b = 456;
    }
    printf("%d", b);

    // prueba 3 - if anidado
    int p = 5;
    int q = 10;
    if (p > 0) {
        if (q > 0) {
            p = p + q;
        } else {
            p = 0;
        }
    }
    printf("%d", p);

    // prueba 4 - while con if dentro
    int w = 10;
    while (w > 0) {
        if (w == 5) {
            a = 1;
        } else {
            a = 0;
        }
        w = w - 1;
    }
    printf("%d", a);

    // prueba 5 - variables locales y globales
    int local = 5;
    local = local + 1;
    global = global + local;
    printf("%d", global);

    // prueba 6 - if con operadores logicos
    int m = 5;
    int n = 10;
    if (m > 0 && n < 20) {
        m = m + 1;
    } else {
        n = n - 1;
    }
    printf("%d", m);

    // prueba 7 - while con variable local acumulando
    int i = 0;
    while (i != 5) {
        i = i + 1;
        a = a + i;
    }
    printf("%d", a);

    // prueba 8 - puts y printf
    puts("resultado:");
    int r = 42;
    printf("%d", r);

    // prueba 9 - operador not con if
    int z = 0;
    if (!z) {
        a = 1;
    } else {
        a = 0;
    }
    printf("%d", a);

    // prueba 10 - combinacion completa while + if + %
    int j = 1;
    int suma = 0;
    while (j <= 10) {
        if (j % 2 == 0) {
            suma = suma + j;
        }
        j = j + 1;
    }
    resultado = suma;
    printf("%d", resultado);
}

(defun factorial (n )
(setq factorial_retorno 0)
(if (= n 1)
(setf factorial_retorno 1)
(setf factorial_retorno (* n (factorial (- n 1) ))))
(princ n)
(princ " ")

(defun fibonacci (n )
(setq fibonacci_retorno 0)
(if (< n 2)
(setf fibonacci_retorno 1)
(setf fibonacci_retorno (+ (fibonacci (- n 1) ) (fibonacci (- n 2) ))))
fibonacci_retorno)

(defun main ()
(setq main_resultado 0)
(setq main_i 0)

(princ main_i)
